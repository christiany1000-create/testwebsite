No videos found. Check API quota or permissions.
'; return; } // Shuffle items slightly so it's a mixed feed items.sort(() => Math.random() - 0.5); items.forEach((item, index) => { if (!item.id || item.id.kind !== 'youtube#video') return; const videoId = item.id.videoId; const channelId = item.snippet.channelId; const title = item.snippet.title; const channelTitle = item.snippet.channelTitle; const autoplayVal = index === 0 ? 1 : 0; const shortHtml = `

${title}

Like
Chat
Sub
`; feed.insertAdjacentHTML('beforeend', shortHtml); }); setupScrollAutoplay(); } catch (err) { console.error('Error loading shorts', err); feed.innerHTML = '
Error loading videos. Check API Key & Quota.
'; } }
